\usepackage[UTF8,fontset=fandol]{ctex}
\usepackage{xcolor}
\usepackage{tikz}
\usetikzlibrary{arrows.meta,calc,fit,positioning}

\definecolor{HwkInk}{HTML}{243247}
\definecolor{HwkLine}{HTML}{718096}
\definecolor{HwkPanel}{HTML}{F4F7FA}
\definecolor{HwkAccent}{HTML}{DCEAF7}

\tikzset{
  hwk title/.style={font=\bfseries\Large,text=HwkInk,align=center},
  hwk participant/.style={draw=HwkLine,fill=white,rounded corners=2pt,
    minimum width=2.6cm,minimum height=0.8cm,font=\small,align=center,text=HwkInk},
  hwk lifeline/.style={draw=HwkLine,densely dashed},
  hwk message/.style={draw=HwkInk,-{Latex[length=2.2mm]},thick},
  hwk return/.style={draw=HwkLine,-{Latex[length=2.2mm]},densely dashed},
  hwk label/.style={fill=white,inner sep=2pt,font=\scriptsize,align=center,text=HwkInk},
  hwk activation/.style={draw=HwkLine,fill=HwkAccent},
  hwk fragment/.style={draw=HwkLine,fill=HwkPanel,fill opacity=.28,text opacity=1},
  hwk fragment label/.style={anchor=north west,font=\bfseries\scriptsize,text=HwkInk}
}

\newcommand{\hwkparticipant}[4]{%
  \node[hwk participant] (#1) at (#2,0) {#3};
  \draw[hwk lifeline] (#1.south) -- ($(#1.south)+(0,-#4)$);
}
\newcommand{\hwkmessage}[4]{%
  \draw[hwk message] ($(#1.south)+(0,-#3)$) --
    node[midway,above,hwk label] {#4} ($(#2.south)+(0,-#3)$);
}
\newcommand{\hwkreturn}[4]{%
  \draw[hwk return] ($(#1.south)+(0,-#3)$) --
    node[midway,above,hwk label] {#4} ($(#2.south)+(0,-#3)$);
}
\newcommand{\hwkactivation}[3]{%
  \draw[hwk activation] ($(#1.south)+(-.09,-#2)$) rectangle ($(#1.south)+(.09,-#3)$);
}
\newcommand{\hwkself}[3]{%
  \draw[hwk message] ($(#1.south)+(0,-#2)$) -- ++(.55,0) -- ++(0,-.38) --
    node[midway,right,hwk label] {#3} ($(#1.south)+(0,-#2-.38)$);
}
